1. 概要と目標

HSP スクリプトからC言語のような構造体を定義し、メンバ名でアクセスできるようにする。 Win32 API の構造体（POINT, RECT, WIN32_FIND_DATA 等）をそのまま定義してDLLに渡せることを目標とする。

目標とする使用例

; 構造体定義
#defstruct POINT
    int x
    int y
#endstruct

; 構造体変数の確保
structdim p, POINT

; メンバアクセス（-> 演算子）
p->x = 100
p->y = 200
mes "x=" + p->x + " y=" + p->y

; Win32 API に渡す
#uselib "user32.dll"
#func GetCursorPos "GetCursorPos" var
GetCursorPos p
mes "cursor: " + p->x + ", " + p->y

3. HSP コンパイラのアーキテクチャ

重要な制約: プリプロセッサとコード生成は別の CToken インスタンスで実行される。 データの受け渡しは出力バッファ（テキスト）とラベルバッファ経由のみ。

3.1 コンパイルの流れ

ソースファイル (.hsp) | v +------------------+ CToken tk1 (インスタンス1) | PreProcess() | - ExpandLine() でマクロ展開 | (token.cpp) | - #const, #define, #module 等を処理 +------------------+ - lb (ラベルバッファ) にシンボル登録 | v 出力バッファ (outbuf) ← プリプロセッサ展開済みテキスト | ※ lb_info として保存可能 v +------------------+ CToken tk2 (インスタンス2) | Compile() | - lb_info からラベル情報を復元 | GenerateCode() | - outbuf のテキストを行単位でパース | (codegen.cpp) | - バイトコード (.ax) を生成 +------------------+ | v バイトコード (.ax ファイル) | v +------------------+ | hsp3.exe | ランタイム | (hsp3code.cpp) | - バイトコードを逐次実行 +------------------+

3.2 データ受け渡しチャネル

チャネル方向内容既存の使用例
出力バッファ (wrtbuf)PP → CG変換済みテキスト行#struct, #deffunc
ラベルバッファ (lb / lb_info)PP → CGシンボル名 → 型/値#const, #define
バイトコード (.ax)CG → RT命令 + パラメータ全コマンド/関数

3.3 既存の #struct の処理パターン

#struct（関数パラメータ用）は以下の方式で PP → CG 間を橋渡ししている:

  1. PP_Struct() (token.cpp): 構造体名+メンバをラベル登録 + 変換済み行を wrtbuf に書き出す
  2. GenerateCodePP_struct() (codegen.cpp): wrtbuf から再パースして STRUCTDAT/STRUCTPRM を構築

この「PP が変換済みテキストを書き出し、CG が再パースする」パターンが HSP コンパイラの標準的な方法。

4. 実装戦略

方針: 既存の #struct パターンに倣い、プリプロセッサが構造体定義を解析して メンバ情報をテキスト形式で出力バッファに書き出す。コード生成フェーズが再パースして使う。

4.1 全体フロー

[HSP ソースコード] #defstruct POINT int x int y #endstruct structdim p, POINT p->x = 100 mes p->x | v プリプロセッサ (token.cpp) [出力バッファに書き出される内容] #defstruct point 8 8 2 ← 名前, total_size, pack, member_count #_smember 0 2 4 0 x ← offset, stype, size, array_count, name #_smember 4 2 4 0 y ← offset, stype, size, array_count, name #endstruct structdim p, 8 ← POINT → サイズ 8 に展開 _struct_poke p, 0, 2, 8, 100 ← p->x = 100 を展開 mes _struct_peek(p, 0, 2, 8) ← p->x を展開 | v コード生成 (codegen.cpp) #defstruct → cg_structdefs テーブル構築 structdim → dim p, 2 (8bytes/4 = 2 int) _struct_poke → バイトコード生成 _struct_peek → バイトコード生成 | v ランタイム (hsp3code.cpp) structdim → int配列確保 + ゼロクリア _struct_poke → メンバ型に応じたバイト書き込み _struct_peek → メンバ型に応じたバイト読み取り → 型付き値を返す

4.2 代替案: -> をプリプロセッサで完全展開

案A: プリプロセッサで展開 採用

; ソース
p->x = 100
mes p->x

; PP 展開後
_struct_poke p, 0, 2, 8, 100
mes _struct_peek(p, 0, 2, 8)
  • PP が構造体情報を持つので解決可能
  • CG は通常のコマンド/関数として処理
  • CG に構造体知識が不要

案B: コード生成で解決 課題あり

; ソース（そのまま CG に渡る）
p->x = 100
mes p->x

; CG が -> を解決
; → cg_structdefs が必要
; → PP と CG が別インスタンスで共有不可
  • PP と CG 間でデータ共有が困難
  • 式中の -> 読み取りが複雑
  • 既存パターンに合わない
案A を採用。-> 演算子の解決をプリプロセッサフェーズで行い、 _struct_poke / _struct_peek() コマンド/関数に展開する。 コード生成フェーズは展開済みのコマンドを通常通り処理するだけ。

5. プリプロセッサフェーズの処理

5.1 #defstruct の処理 (Preprocess 内)

// token.cpp - Preprocess() 内

if (tstrcmp(word, "defstruct") || tstrcmp(word, "defunion")) {
    // 1. 構造体名を取得
    // 2. pp_defstruct_level++ でメンバ行スキップモード開始
    // 3. StructDef を作成して pp_structdefs に push
    // 4. wrtbuf に "#defstruct name size pack member_count" を書き出す（CG 用）
    return PPRESULT_WROTE_LINE;
}

5.2 メンバ行の処理 (ExpandLine 内)

// token.cpp - ExpandLine() 内

if (pp_defstruct_level > 0 && !is_preprocess_line) {
    // 1. メンバ行をパース: "int x" → type=SMT_INT, name="x", offset=0, size=4
    // 2. StructDef.members に追加
    // 3. wrtbuf に "#_smember offset type size array_count name" を書き出す
    buf->PutStr(formatted_member_line);
    buf->PutCR();
    pline++;
    continue;
}

5.3 #endstruct の処理

// token.cpp - Preprocess() 内

if (tstrcmp(word, "endstruct") || tstrcmp(word, "endunion")) {
    // 1. total_size を確定（アライメント切り上げ）
    // 2. 構造体名を LAB_TYPE_PPVAL 定数として登録（値 = 0x7F000000 | struct_id）
    //    → structdim p, POINT の POINT がこの値に展開される
    // 3. wrtbuf に "#endstruct" を書き出す
    return PPRESULT_WROTE_LINE;
}

5.4 -> 演算子の展開 (ExpandTokens 内)

ここが最も重要な部分。プリプロセッサのトークン展開時に var->member パターンを検出し、 _struct_poke / _struct_peek() に変換する。

書き込みパターン: p->x = 100

// token.cpp - ExpandTokens() 内でトークンストリームを走査

// 入力トークン列: [p] [->] [x] [=] [100]
//
// 検出条件:
//   1. 変数トークン (var) の後に -> が続く
//   2. var が structdim で登録された構造体変数
//
// 変換:
//   → _struct_poke p, 0, 2, 8, 100
//     引数: var, member_offset, member_type, struct_size, value_expression
//
// 配列の場合: pts(3)->x = 100
//   → _struct_poke pts(3), 0, 2, 8, 100
//     ※ ランタイムが pts(3) の配列オフセットを考慮する

読み取りパターン: mes p->x

// 入力トークン列: [mes] [p] [->] [x]
//
// 変換:
//   → mes _struct_peek(p, 0, 2, 8)
//     引数: var, member_offset, member_type, struct_size
//     戻り値: member_type に応じた型付き値 (int/int64/double/str)

判定に必要な情報

情報格納場所取得方法
変数が構造体かpp_var_structid マップstructdim 行の処理時に登録
構造体の定義pp_structdefs ベクタ#defstruct 処理時に構築

// 入力: structdim p, POINT

// PP 処理:
//   1. POINT は LAB_TYPE_PPVAL 定数として展開済み → 0x7F000000 | struct_id
//   2. しかし _struct_poke 展開のため、変数→構造体マッピングも必要
//   3. ExpandTokens で "structdim" の後の変数名と構造体名を検出
//   4. pp_var_structid[変数名] = struct_id を記録

// 出力: structdim p, 8
//   ※ 0x7F000000|id ではなく、直接サイズに展開する方がシンプル

7. ランタイムの処理

7.1 structdim (PROGCMD 0x22)

// hsp3code.cpp

case 0x22:  // structdim var, total_size [, count]
{
    PVal *pval = code_getpval();
    int struct_size = code_getdi(0);
    int count = code_getdi(1);
    int total_bytes = struct_size * count;
    int dim_count = (total_bytes + 3) / 4;

    // int 配列として確保
    HspVarCoreDimWC(pval, HSPVAR_FLAG_INT, dim_count, 0, 0, 0);
    // ゼロクリア
    memset(pval->pt, 0, dim_count * sizeof(int));
    break;
}

7.2 _struct_poke (PROGCMD 0x23)

// hsp3code.cpp

case 0x23:  // _struct_poke var, member_offset, member_type, struct_size, value
{
    PVal *pval;
    APTR aptr = code_getva(&pval);  // 配列インデックスも考慮
    int member_offset = code_geti();
    int member_type = code_geti();
    int struct_size = code_geti();

    // 配列要素のベースオフセット計算
    // aptr は int 配列のインデックスなので byte_offset = aptr * 4
    // ※ structdim pts, POINT, 100 の場合:
    //   pts(3)->x → aptr=3, struct_size=8
    //   base = pval->pt + (aptr * struct_size) + member_offset
    char *base = (char*)pval->pt + (aptr * struct_size) + member_offset;

    // 型に応じた書き込み
    switch (member_type) {
    case SMT_BYTE:   *base = (char)code_geti(); break;
    case SMT_SHORT:  *(short*)base = (short)code_geti(); break;
    case SMT_INT:    *(int*)base = code_geti(); break;
    case SMT_INT64:  *(int64_t*)base = code_geti64(); break;
    case SMT_FLOAT:  *(float*)base = (float)code_getdd(0.0); break;
    case SMT_DOUBLE: *(double*)base = code_getdd(0.0); break;
    case SMT_PTR:    *(void**)base = (void*)(intptr_t)code_geti64(); break;
    case SMT_BOOL:   *(int*)base = code_geti(); break;
    case SMT_CHAR_ARRAY: {
        char *src = code_gets();
        strncpy(base, src, /* member_size */);
        break;
    }
    // ...
    }
    break;
}

7.3 _struct_peek (INTFUNC 0x17)

// hsp3int.cpp

case 0x17:  // _struct_peek(var, member_offset, member_type, struct_size)
{
    PVal *pval;
    int size;
    char *sptr = code_getvptr(&pval, &size);
    int member_offset = code_geti();
    int member_type = code_geti();
    int struct_size = code_geti();

    char *base = sptr + member_offset;

    switch (member_type) {
    case SMT_INT:
        reffunc_intfunc_ivalue = *(int*)base;
        *type_res = HSPVAR_FLAG_INT;
        ptr = &reffunc_intfunc_ivalue;
        break;
    case SMT_DOUBLE:
        reffunc_intfunc_dvalue = *(double*)base;
        *type_res = HSPVAR_FLAG_DOUBLE;
        ptr = &reffunc_intfunc_dvalue;
        break;
    case SMT_CHAR_ARRAY:
        *type_res = HSPVAR_FLAG_STR;
        ptr = base;
        break;
    // ...
    }
    break;
}

8. #func / #cfunc での構造体パラメータ

8.1 ポインタ渡し（最もシンプル）

; 構造体のバッファポインタを渡す

構造体変数は int 配列なので、既存の var パラメータタイプで参照渡しが可能。 DLL 側には配列のポインタ（= 構造体のバイトバッファポインタ）が渡される。

9. 判明した課題と解決策

課題1: PP と CG が別インスタンス 解決済み

問題: cg_structdefs がプリプロセッサとコード生成で共有されない。

解決: PP が -> を完全展開して _struct_poke / _struct_peek() に変換。 CG は構造体情報を知る必要がない。

課題2: -> の展開タイミング

問題: -> 演算子はプリプロセッサの ExpandTokens で展開する必要がある。 しかし ExpandTokens はトークン単位の置換で、文脈依存の変換は難しい。

解決案A: ExpandTokens の後、行単位で -> パターンを検出して変換する後処理を追加。

解決案B: SendLineBuf の段階で行テキストを直接書き換える。

解決案C: ExpandTokens 内でトークン 0x65 (->)を検出して変換。

課題3: 配列構造体のオフセット計算

問題: pts(3)->x の場合、ランタイムで aptr * struct_size + member_offset を計算する必要がある。 しかし code_getva は int 配列のインデックスを返すので、構造体サイズ分のスケーリングが必要。

解決: PP 展開時に _struct_poke pts(3), 0, 2, 8, 100 とし、 ランタイムの _struct_poke が aptr * struct_size + member_offset で計算。 ただし code_getva が返す aptr は int 配列インデックス（= 要素番号）なので、 structdim pts, POINT, 100 で dim pts, 200 (= 100 * 8 / 4) になった場合、 pts(3) の aptr=3 は int 配列の 3 番目（= 12 バイト目）を指す。 これは POINT 構造体の 1.5 個目に相当し、正しくない。

修正案: 構造体配列は PP が pts(3) を pts(3 * struct_size / 4) に展開するか、 ランタイムで struct_size を使って aptr をスケーリングする。

課題4: 式中での -> 読み取り

問題: a = p->x + p->y のように式の途中で -> が出てくる場合の処理。

解決: PP が _struct_peek(p, 0, 2, 8) + _struct_peek(p, 4, 2, 8) に展開するので問題なし。 PP での展開が正しく行われれば、CG は通常の式として処理できる。
